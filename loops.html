Question 1: Enter a number from the user and print the series from 1 to that number ?

Program:

#include <stdio.h>

void main() {
    int n, i;

    printf("Enter a number: ");
    scanf("%d", &n);

    for (i = 1; i <= 9; i++) {
        printf("%d ", i);
    }
}

Input:

Enter numbers: 9

Output:

1 2 3 4 5 6 7 8 9

Question 5: Enter a number from the user and find out the binary of that number ?

Program:

#include <stdio.h>

void main()
{
    int n, rem, binary = 0, place = 1;

    printf("Enter a number: ");
    scanf("%d", &n);

    while(n > 0)
    {
        rem = n % 2;
        binary = binary + rem * place;
        place = place * 10;
        n = n / 2;
    }

    printf("Binary = %d", binary);
}

Input:

Enter numbers: 10

Output:

Binary;1010
