Question 1: Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

Program:

#include <stdio.h>
struct employee
{
    int id;
    char name[50];
    char designation[50];
};

void main()
{
    struct employee e[50];
    int i;

    for(i = 0; i < 50; i++)
    {
        printf("\nEnter details of employee %d\n", i + 1);

        printf("Enter Employee ID: ");
        scanf("%d", &e[i].id);

        printf("Enter Name: ");
        scanf("%s", e[i].name);

        printf("Enter Designation: ");
        scanf("%s", e[i].designation);
    }

    printf("\nEmployee Details\n");
    printf("ID\tName\tDesignation\n");

    for(i = 0; i < 50; i++)
    {
        printf("%d\t%s\t%s\n", e[i].id, e[i].name, e[i].designation);
    }

}

Question 2: Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

Program:

#include <stdio.h>

struct student
{
    int roll;
    char name[50];
    float fee;
    char dob[20];
};

VOID main()
{
    struct student s[50], temp;
    int i, j, n;

    printf("Enter number of students: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nEnter details of student %d\n", i + 1);

        printf("Enter Roll Number: ");
        scanf("%d", &s[i].roll);

        printf("Enter Name: ");
        scanf("%s", s[i].name);

        printf("Enter Fee: ");
        scanf("%f", &s[i].fee);

        printf("Enter DOB (DD/MM/YYYY): ");
        scanf("%s", s[i].dob);
    }

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(s[j].roll > s[j + 1].roll)
            {
                temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }
        }
    }

    printf("\nStudent Details in Ascending Order\n");
    printf("Roll\tName\tFee\tDOB\n");

   for(i = 0; i < n; i++)
   {
   printf("%d\t%s\t%.2f\t%s\n",
   s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }

}
