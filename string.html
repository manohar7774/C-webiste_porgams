Question 1: Enter a string from user and convert in uppercase ?

Program:

#include <stdio.h>

void main() {
    char str[100];
    int i;

    printf("Enter a string: ");
    scanf("%s",str);

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] >= 'a' && str[i] <= 'z') {
            str[i] = str[i] - 32;
        }
    }

    printf("Uppercase string = %s", str);
}

Input:

Enter a string:good morning

Output:

GOOD MORNING

Question 2: Enter a string from user and convert into lowercase ?

Program:

#include <stdio.h>

void main() {
    char str[100];
    int i;

    printf("Enter a string: ");
    scanf("%s", str);

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] >= 'A' && str[i] <= 'Z') {
            str[i] = str[i] + 32;
        }
    }

    printf("Lowercase string = %s", str);
}

Input:

Enter a string: GOOD BYE

Output:

good bye

Question 3: Enter the data from user the data consist of roll no. ,name ,class , marks of 5 subjects ?

Program:

#include <stdio.h>
void main()
{
    int roll, i;
    char name[50], class[20];
    float marks[5];

    printf("Enter Roll Number: ");
    scanf("%d", &roll);

    printf("Enter Name: ");
    scanf("%s", name);

    printf("Enter Class: ");
    scanf("%s", class);

    printf("Enter marks of 5 subjects:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%f", &marks[i]);
    }

    printf("\nStudent Details\n");
    printf("Roll Number: %d\n", roll);
    printf("Name: %s\n", name);
    printf("Class: %s\n", class);

    printf("Marks:\n");

    for(i = 0; i < 5; i++)
    {
        printf("Subject %d: %.2f\n", i + 1, marks[i]);
    }


}

Input:

Enter Roll Number: 101
Enter Name: Rahul
Enter Class: BCA
Enter marks of 5 subjects:
80
75
90
85
88

Output:

Student Details
Roll Number: 101
Name: XYZ
Class: BCA
Marks:
Subject 1: 80
Subject 2: 75
Subject 3: 90
Subject 4: 85
Subject 5: 88
