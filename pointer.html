Question 1: Display the value of a variable using a pointer ?

Program:

#include <stdio.h>
void main()
{
    int a = 10;
    int *p;

    p = &a;

    printf("%d", *p);
}

Output:

10

Question 2: Add two numbers using pointers ?

Program:

#include <stdio.h>
void main()
{
    int a, b, sum;
    int *p, *q;

    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);

    p = &a;
    q = &b;

    sum = *p + *q;

    printf("Sum = %d", sum);
}

Question 3: Swap two numbers using pointers?

Program:

#include <stdio.h>
void main()
{
    int a, b, temp;
    int *p, *q;

    printf("Enter two numbers: ");
    scanf("%d", &a, &b);

    p = &a;
    q = &b;

    temp = *p;
    *p = *q;
    *q = temp;

    printf("After swapping:\n");
    printf("a = %d", a);
    printf("b = %d", b);
}

Question 6: Find the sum of array elements using pointers?

Program:

#include <stdio.h>
void main()
{
    int a[5], i, sum = 0;
    int *p;

    printf("Enter 5 elements: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    p = a;

    for(i = 0; i < 5; i++)
    {
        sum = sum + *(p+i);
    }

    printf("Sum = %d", sum);
}

Input:

Enter 5 elements: 44 55 66 77 88

Output:

sum:330
