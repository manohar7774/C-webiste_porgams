Question 1: Write a program to enter the number from the user and check it is positive or negative or 0 (zero) ?

Program:

#include <stdio.h>

int main()
{
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);

    if(num > 0)
    {
        printf("The number is positive.");
    }
    else if(num < 0)
    {
        printf("The number is negative.");
    }
    else
    {
        printf("The number is zero.");
    }

    return 0;
}

Input:

Enter the numbers: -7

Output:

Number is negative

Question 2:Enter a character from the user and check that it is an alphabet ?

Program:

#include <stdio.h>

void main()
{
    char ch;

    printf("Enter a character: ");
    scanf("%c", &ch);

    if((ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z'))
    {
        printf("The entered character is an alphabet.");
    }
    else
    {
        printf("The entered character is not an alphabet.");
    }

Input:

Enter a Character: M

Output:

character is alphbet
