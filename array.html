Question 1: Find the Sum of All Elements in a Array?

Program:

#include <stdio.h>

void main() {

    int arr[5] = {10, 20, 30, 40, 50};
    int sum = 0;

    for (int i = 0; i < 5; i++) {
        sum=sum+arr[i];
    }

    printf("The sum of all elements in the array is: %d\n", sum);

}

Input:

Enter numbers: 10 20 30 40 50

Output:

The sum of all elements in the array is: 150

Question 2:Find the Sum of All Elements in a 2D Matrix

Program:

#include <stdio.h>

void main() {
    int matrix[3][3] = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    int sum = 0;

    printf("Matrix grid:\n");

    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            printf("%d ", matrix[i][j]);
            sum += matrix[i][j];
        }
        printf("\n");
    }
    printf("%d\n\n", sum);

}

Input:

Enter a number:1 2 3 4 5 6 7 8 9

Output:

45

Question 3:Enter an array from user and print the array in reverse order?

Program:

#include <stdio.h>

int main()
{
    int arr[100], num, i;

    printf("Enter the number of elements: ");
    scanf("%d", &num);

    for(i = 0; i < num; i++)
    {
        printf("Enter element %d: ", i + 1);
        scanf("%d", &arr[i]);
    }

    printf("\nArray in reverse order:\n");

    for(i = num - 1; i >= 0; i--)
    {
        printf("%d ", arr[i]);
    }
    return 0;
}

Input:

Enter the number of elements: 5
Enter element 1: 10
Enter element 2: 20
Enter element 3: 30
Enter element 4: 40
Enter element 5: 50

Output:

Array in reverse: 50 40 30 20 10
