Question 1:Write a program to enter 10 numbers from the user, add the numbers, and display the answer.

Program:

#include <stdio.h>

void main() {

    {
    int i, num, sum = 0;

    for(i = 1; i <= 10; i++)
    {
        printf("Enter number %d: ", i);
        scanf("%d", &num);
        sum = sum + num;
    }

    printf("Sum of 10 numbers = %d", sum);
}

Input:

Enter number 1: 1
Enter number 2: 2
Enter number 3: 3
Enter number 4: 4
Enter number 5: 5
Enter number 6: 6
Enter number 7: 7
Enter number 8: 8
Enter number 9: 9
Enter number 10: 10

Output:

The sum of all numbers is:55

Question 2:Write a C program to swap two numbers using a function

Program:

#include <stdio.h>

void main() {
    int temp;
    temp = *a;
    *a = *b;
    *b = temp;
}

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    printf("Before swapping: a = %d, b = %d\n", a, b);

    swap(&a, &b);

    printf("After swapping: a = %d, b = %d\n", a, b);
}

Input:

Enter two numbers: 10 20

Output:

Before swaping = 10, b = 20
After swapping: a = 20, b = 10

Question 3:Swap numbers using Call by function ?

Program:

#include <stdio.h>
void swap(int *num1, int *num2)
{
    int temp;

    temp = *num1;
    *num1 = *num2;
    *num2 = temp;
}

int main()
{
    int num1, num2;

    printf("Enter number 1: ");
    scanf("%d", &num1);

    printf("Enter number 2: ");
    scanf("%d", &num2);

    printf("Number 1 = %d\n", num1);
    printf("Number 2 = %d\n", num2);

    swap(&num1, &num2);

    printf("Number 1 = %d\n", num1);
    printf("Number 2 = %d\n", num2);

}
}

Input:

Enter two numbers: 7 6

Output:

Swaped numbers: 6 7
