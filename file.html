Question 1: Create a file and write a quote into it ?

Program:

#include <stdio.h>

void main()
{
    FILE *fp;
    fp = fopen("name.txt", "w");

    fprintf(fp, "GOOD MORNING");

    fclose(fp);
}

Question 2:Read and display the contents of a file?

Program:

#include <stdio.h>

void main()
{
    FILE *fp;
    char ch;

    fp = fopen("name.txt", "r");

    while((ch = fgetc(fp)) != EOF)
    {
        printf("%c", ch);
    }

    fclose(fp);
}

Question 3: Write 10 numbers into a file and display them ?

Program:

#include <stdio.h>
void main()
{
    FILE *fp;
    int i, n;

    fp = fopen("number.txt", "w");

    for(i = 1; i <= 5; i++)
    {
        scanf("%d", &n);
        fprintf(fp, "%d ", n);
    }

    fclose(fp);

    fp = fopen("number.txt", "r");

    while(fscanf(fp, "%d", &n) != EOF)
    {
        printf("%d\n", n);
    }

    fclose(fp);
}

Question 4: Count the number of characters in a file

Program:

#include <stdio.h>
     void main()
{
    FILE *fp;
    char ch;
    int count = 0;

    fp = fopen("name.txt", "r");

    while((ch = fgetc(fp)) != EOF)
    {
        count++;
    }

    printf("no of characters = %d", count);

    fclose(fp);
}

Question 5: Create a File and Count the Number of Words in It

Program:

#include <stdio.h>
   void main()
{
    FILE *fp;
    char ch;
    int count = 0, flag = 0;

    fp = fopen("data.txt", "w");

    fprintf(fp, "Welcome to C programming language");

    fclose(fp);

    fp = fopen("data.txt", "r");

    while((ch = fgetc(fp)) != EOF)
    {
        if(ch == ' ' || ch == '\n' || ch == '\t')
        {
            flag = 0;
        }
        else if(flag == 0)
        {
            count++;
            flag = 1;
        }
    }

    fclose(fp);

    printf("Total number of words = %d", count);
}
